# ============================================
# GALAXY S24 ULTRA - ONEUI 9 (ANDROID 17)
# ============================================
# Package list captured from a Galaxy S24 Ultra (SM-S928B, CSC CHO,
# Chile open) running OneUI 9.0 / Android 17, October 2026 security patch
# (build S928BXXS6EZJ1).
#
# This file adds the packages that the 01-08 lists do not cover on this
# device. The 01-07 lists still apply; 08-z-fold8.txt covers the shared
# One UI 9 platform extras (Now Brief, Linux terminal, XR glasses stub and
# the rest). Every entry in 08 was re-checked on this device: everything it
# names is present here except com.samsung.android.coldwalletservice and
# com.samsung.SMT.lang_pt_br_l01.
#
# HOW THIS LIST WAS BUILT: every package below was confirmed present on the
# device and inspected with dumpsys and with its APK manifest (label,
# launcher activity, shared UID, privileged flags) to determine what it does.
# These removals were NOT test-run on the reference device. Start with the
# SAFE section, reboot, and check the features you use.
#
# ANDROID 17 / ONEUI 9: on this build the shell cannot uninstall a system
# app for a single user. `pm uninstall --user 0` fails with
# "Failure [only root can delete system app for a particular user]"
# (verified on the reference device: 25 of 25 packages refused). Every
# command below therefore uses `pm disable-user --user 0`, which stops the
# app for user 0 and is reversible with `pm enable <package>`. Storage is
# not freed: the packages stay on the system partition.
#
# VARIANT NOTE: this is an international SM-S928B on a Latin America CSC.
# TTS packs and carrier apps differ by region, and the US SM-S928U/U1 ships
# extra carrier apps that are not listed here. Missing packages just report
# "not installed" and are safe to ignore.
#
# S24 ULTRA DIFFERENCES FROM THE FOLD 8 LIST: this phone has no cover
# screen, no Flex mode, and no satellite packages. It does have an S Pen, a
# full eSIM stack and UWB, which the Fold 8 list does not document. Read the
# S PEN and KEEP sections before removing anything.
#
# Every change is reversible with `pm enable <package>` (RESTORE-GUIDE.md
# covers the uninstall path on older builds).
# ============================================


# ============================================
# SAFE TO REMOVE - SAMSUNG
# ============================================

# Wi-Fi Tips (suggests open and high-quality public networks)
adb shell pm disable-user --user 0 com.samsung.android.net.wifi.wifiguider

# Shortcut backup and restore (backs up home screen shortcuts to Samsung Cloud)
adb shell pm disable-user --user 0 com.samsung.android.shortcutbackupservice

# Network diagnostics reporter
adb shell pm disable-user --user 0 com.samsung.android.networkdiagnostic


# ============================================
# SAFE TO REMOVE - GOOGLE / ANDROID
# ============================================

# SIM Toolkit (legacy carrier SIM menu apps). Chilean carriers do not use
# them. Keep if your carrier ships a SIM menu you actually open.
adb shell pm disable-user --user 0 com.android.stk

# SIM Toolkit for the second SIM slot
adb shell pm disable-user --user 0 com.android.stk2

# SIM app dialog (the prompt STK uses to ask before opening a SIM app)
adb shell pm disable-user --user 0 com.android.simappdialog

# OPTIONAL - FEATURE TRADE-OFFS

# Single Take (camera mode that captures several photos and videos at once)
adb shell pm disable-user --user 0 com.samsung.android.singletake.service

# Motion photo (captures a short video with each still)
adb shell pm disable-user --user 0 com.samsung.android.motionphoto.app

# Camera and Gallery filter packs
adb shell pm disable-user --user 0 com.samsung.android.provider.filterprovider

# Video call effects (background blur and colour correction in third-party
# video call apps)
adb shell pm disable-user --user 0 com.samsung.android.vtcamerasettings

# Gallery Cloud Sync (OneDrive sync of photos and albums). Keep if you back
# up your Gallery to OneDrive.
adb shell pm disable-user --user 0 com.samsung.android.agent.storage

# Smart View (screen mirroring to a TV or monitor)
adb shell pm disable-user --user 0 com.samsung.android.smartmirroring

# Audio mirroring (casts audio to a TV or speaker)
adb shell pm disable-user --user 0 com.samsung.android.audiomirroring

# Nearby media sharing (DLNA streaming from Gallery and Video to a TV)
adb shell pm disable-user --user 0 com.samsung.android.allshare.service.mediashare

# SmartThings Companion (device pairing helpers for SmartThings)
adb shell pm disable-user --user 0 com.samsung.android.service.stplatform

# App protection (McAfee-powered scan of installed apps). This is the
# Settings > Security and privacy > App protection feature, not Play Protect.
adb shell pm disable-user --user 0 com.samsung.android.sm.devicesecurity

# Digital Wellbeing (screen time, app timers, focus mode)
adb shell pm disable-user --user 0 com.samsung.android.forest

# Samsung Video Player (the stock player; Gallery has its own playback)
adb shell pm disable-user --user 0 com.samsung.android.video

# One-handed mode
adb shell pm disable-user --user 0 com.sec.android.easyonehand

# Adapt Sound (hearing profile for headphones)
adb shell pm disable-user --user 0 com.sec.hearingadjust

# Sound quality and effects (Dolby Atmos and the equalizer)
adb shell pm disable-user --user 0 com.sec.android.app.soundalive

# Link to Windows Service (Phone Link and app streaming on a Windows PC)
adb shell pm disable-user --user 0 com.samsung.android.mdx

# Storage share (use this phone's storage from another Galaxy device or PC)
adb shell pm disable-user --user 0 com.samsung.android.hwresourceshare.storage

# Galaxy Resource Updater (updates Samsung app resource and asset packs).
# Apps keep working with their bundled resources.
adb shell pm disable-user --user 0 com.samsung.android.gru

# Google Wi-Fi Provisioner (carrier Wi-Fi auto-connect and Passpoint setup)
adb shell pm disable-user --user 0 com.google.android.apps.carrier.carrierwifi


# ============================================
# S PEN - REMOVE ONLY IF YOU NEVER USE THE S PEN
# ============================================
# These control S Pen features. Removing them does not break the touchscreen,
# but it disables the matching S Pen functionality.

# Air command (the floating S Pen menu, Smart select, Screen write shortcuts)
adb shell pm disable-user --user 0 com.samsung.android.service.aircommand

# Air command manager (the system-side manager for the Air command menu)
adb shell pm disable-user --user 0 com.samsung.android.aircommandmanager

# Magnify (the S Pen magnifier and its accessibility entry point)
adb shell pm disable-user --user 0 com.samsung.android.app.readingglass

# Handwriting service (S Pen handwriting recognition, also used by the
# handwriting mode of Samsung Keyboard and by Samsung Notes)
adb shell pm disable-user --user 0 com.samsung.android.sdk.handwriting


# ============================================
# CAUTION - DIAGNOSTIC AND SYSTEM AGENTS
# ============================================
# These were inspected on the reference device and hold no user-facing
# feature, but they run under the system UID and ship diagnostics. Remove
# only if you accept that Samsung Members and support cannot pull logs.

# Samsung diagnostic log agent (used by Samsung Members and service support)
adb shell pm disable-user --user 0 com.sec.android.diagmonagent

# Device Quality Agent (Samsung quality and error telemetry)
adb shell pm disable-user --user 0 com.samsung.android.dqagent

# Device care (Settings > Battery and device care, storage and memory
# management, optimisation). Removing it also removes the battery usage UI
# entry point.
adb shell pm disable-user --user 0 com.samsung.android.lool
